const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const path = require('path'); const fs = require('fs'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*", methods: ["GET", "POST"] } }); app.get('/', (req, res) => { const Paths = [ path.join(__dirname, 'index.html'), path.join(__dirname, 'public', 'index.html'), path.join(__dirname, 'src', 'index.html') ]; for (const FilePath of Paths) { if (fs.existsSync(FilePath)) { return res.sendFile(FilePath); } } const FindIndexHtml = (Dir) => { let Entries; try { Entries = fs.readdirSync(Dir, { withFileTypes: true }); } catch { return null; } for (const Entry of Entries) { const FullPath = path.join(Dir, Entry.name); if (Entry.isFile() && Entry.name.toLowerCase() === 'index.html') { return FullPath; } if (Entry.isDirectory() && Entry.name !== 'node_modules' && Entry.name !== '.git') { const Found = FindIndexHtml(FullPath); if (Found) return Found; } } return null; }; const FoundPath = FindIndexHtml(__dirname); if (FoundPath) { return res.sendFile(FoundPath); } res.status(404).send('index.html not found'); }); let players = {}; function ClonePlayers() { const Out = {}; for (const id in players) { Out[id] = { id: players[id].id, x: players[id].x, y: players[id].y, name: players[id].name, score: players[id].score, isDead: players[id].isDead }; } return Out; } function NormalizeName(Name) { if (typeof Name !== 'string') return 'Duck'; const Clean = Name.trim(); if (!Clean) return 'Duck'; return Clean.slice(0, 10); } function EnsurePlayer(socketId, data) { const Name = NormalizeName(data && data.name); const X = typeof data?.x === 'number' ? data.x : 100; const Y = typeof data?.y === 'number' ? data.y : 200; if (!players[socketId]) { players[socketId] = { id: socketId, x: X, y: Y, name: Name, score: 0, isDead: false }; return true; } players[socketId].x = X; players[socketId].y = Y; players[socketId].name = Name || players[socketId].name || 'Duck'; if (typeof data?.score === 'number') players[socketId].score = data.score; if (typeof data?.isDead === 'boolean') players[socketId].isDead = data.isDead; return false; } io.on('connection', (socket) => { socket.on('join-game', (data = {}) => { const IsNew = EnsurePlayer(socket.id, data); socket.emit('current-players', ClonePlayers()); if (IsNew) { socket.broadcast.emit('player-joined', { id: players[socket.id].id, x: players[socket.id].x, y: players[socket.id].y, name: players[socket.id].name, score: players[socket.id].score, isDead: players[socket.id].isDead }); } else { socket.broadcast.emit('player-state', { id: players[socket.id].id, x: players[socket.id].x, y: players[socket.id].y, name: players[socket.id].name, score: players[socket.id].score, isDead: players[socket.id].isDead }); } }); socket.on('update-position', (data = {}) => { if (!players[socket.id]) { players[socket.id] = { id: socket.id, x: 100, y: 200, name: 'Duck', score: 0, isDead: false }; } if (typeof data.x === 'number') players[socket.id].x = data.x; if (typeof data.y === 'number') players[socket.id].y = data.y; if (typeof data.score === 'number') players[socket.id].score = data.score; if (typeof data.isDead === 'boolean') players[socket.id].isDead = data.isDead; io.emit('player-state', { id: players[socket.id].id, x: players[socket.id].x, y: players[socket.id].y, name: players[socket.id].name, score: players[socket.id].score, isDead: players[socket.id].isDead }); }); socket.on('disconnect', () => { if (players[socket.id]) { delete players[socket.id]; io.emit('player-disconnected', socket.id); } }); }); const PORT = process.env.PORT || 8080; server.listen(PORT, '0.0.0.0', () => { console.log(`Socket.io server running on port ${PORT}`); });
